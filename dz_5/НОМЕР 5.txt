NOMINALY = [5000, 2000, 1000, 500, 200, 100]

summa = int(input("Введите сумму для снятия (кратна 100): "))

if summa % 100 != 0:
    print("Ошибка: сумма должна быть кратна 100.")
else:
    ostatok = summa
    for note in NOMINALY:
        count = ostatok // note
        if count:
            print(f"{note} руб. × {count} = {note * count} руб.")
            ostatok -= count * note
