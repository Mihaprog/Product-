temp = float(input("Температура: "))
davlenie = int(input("Давление: "))
pulse = int(input("Пульс: "))

if temp < 35 or temp > 38 or davlenie < 105 or davlenie > 140 or pulse < 55 or pulse > 110:
    print("Требуется врач")

elif (35 <= temp < 36 or 37 < temp <= 38) or (105 <= davlenie < 110 or 130 < davlenie <= 140) or (55 <= pulse < 60 or 100 < pulse <= 110):
    print("Легкое недомогание")
else:
    print("Нормальное состояние")
