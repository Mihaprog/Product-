x1, y1, x2, y2 = map(int, input("Введите x1 y1 x2 y2: ").split())

# Логика: на шахматной доске клетки чередуются.
# Если сумма координат чётная — клетка одного цвета,  если нечётная — другого.
# Значит, две клетки одного цвета, если чётность их сумм совпадает.

if (x1 + y1) % 2 == (x2 + y2) % 2:
    print("ДА")  # Одинаковый
else:
    print("НЕТ")   # Разный
