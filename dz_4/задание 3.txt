num = int(input("Введите положительное четырёхзначное число: "))
tis = num // 1000
sot = (num // 100) % 10
des = (num // 10) % 10
edin = num % 10

print("Цифра в позиции тысяч равна", tis)
print("Цифра в позиции сотен равна", sot)
print("Цифра в позиции десятков равна", des)
print("Цифра в позиции единиц равна", edin)
